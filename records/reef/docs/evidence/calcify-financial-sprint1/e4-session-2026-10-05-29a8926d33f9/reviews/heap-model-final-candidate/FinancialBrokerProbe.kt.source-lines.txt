385:     internal class OutputVerifier(private val config: JsonNode) {
386:         val owners = linkedMapOf<String, ObjectNode>()
387:         val histories = linkedMapOf<String, MutableList<JsonNode>>()
388:         private val oracles = mutableMapOf<String, FinancialOracle>()
389:         private data class Staged(val input: JsonNode, val digest: String, val position: Long)
390:         private val pending = mutableMapOf<String, LinkedHashMap<String, Staged>>()
391:         private val nextDelivery = mutableMapOf<String, Long>()
392:         private val cursor = mutableMapOf<String, Long>()
393:         var business = 0; private set
394:         var settled = 0; private set
395:         private var ordinal = -1L
396:         fun accept(output: JsonNode) {
397:             require(output["inputOrdinal"].asLong() == ordinal + 1) { "output logical prefix gap" }; ordinal++
398:             val expectedEnvelope = config["inputs"][ordinal.toInt()]
399:             val domain = expectedEnvelope["domain"].asText()
400:             require(output["domain"].asText() == domain) { "cross-domain output" }
401:             val oracle = oracles.getOrPut(domain) { FinancialOracle(config["genesis"][domain]["balances"], config["policy"]) }
402:             val queue = pending.getOrPut(domain) { linkedMapOf() }
403:             val beforeSeq = oracle.businessView()["businessSeq"].asLong()
404:             var disposition: String? = null
405:             fun enqueue(input: JsonNode) {
406:                 val digest = requestDigest(input)
407:                 val key = json.writeValueAsString(listOf(actionKey(input), digest))
408:                 if (!queue.containsKey(key)) {
409:                     val position = nextDelivery[domain] ?: 0L
410:                     queue[key] = Staged(input, digest, position); nextDelivery[domain] = position + 1
411:                 }
412:             }
413:             when (expectedEnvelope["mode"].asText()) {
414:                 "STAGE" -> { enqueue(expectedEnvelope["input"]); disposition = "STAGED" }
415:                 "EXECUTE" -> {
416:                     disposition = oracle.execute(expectedEnvelope["input"])["disposition"].asText()
417:                     if (disposition == "STAGED") enqueue(expectedEnvelope["input"])
418:                 }
419:                 "DRAIN" -> {
420:                     val view = oracle.businessView()
421:                     val item = if (view["continuation"].isNull) queue.entries.minByOrNull { it.value.position } else queue.entries.filter {
422:                         it.value.input.path("kind").asText().uppercase() == "CONTINUE" && it.value.input.path("payload").path("workId").asText() == view["dueWork"][0].asText()
423:                     }.minByOrNull { it.value.position }
424:                     if (item != null) {
425:                         queue.remove(item.key); cursor[domain] = (cursor[domain] ?: 0L) + 1
426:                         disposition = oracle.execute(item.value.input)["disposition"].asText()
427:                     }
428:                 }
429:                 else -> error("unsupported source dispatch")
430:             }
431:             val owner = owners.getOrPut(domain) { emptyOwner() }
432:             val domainHistory = histories.getOrPut(domain) { mutableListOf() }
433:             var emittedBusiness = 0
434:             output["records"].forEach { history ->
435:                 verifyRecord(history, domainHistory.lastOrNull()); domainHistory.add(history)
436:                 history["changes"].forEach { change -> apply(owner, change) }
437:                 if (history["kind"].asText() == "BUSINESS") emittedBusiness++
438:             }
439:             require(domainHistory.isNotEmpty() && domainHistory.first()["kind"].asText() == "GENESIS") { "missing genesis authority" }
440:             require(emittedBusiness.toLong() == oracle.businessView()["businessSeq"].asLong() - beforeSeq) { "missing or extra business authority" }
441:             require(output["disposition"]?.takeUnless { it.isNull }?.asText() == disposition) { "source disposition mismatch" }
442:             oracle.assertMatches(businessView(owner), "broker domain=$domain input=$ordinal")
443:             val expectedStaging = json.valueToTree<JsonNode>(queue.mapValues { (_, value) -> mapOf("input" to value.input, "digest" to value.digest, "position" to value.position) })
444:             require(canonical(owner["stagedInputs"]) == canonical(expectedStaging)) { "missing or extra staged authority" }
445:             require(owner["nextDelivery"].asLong() == (nextDelivery[domain] ?: 0L) && owner["deliveryCursor"].asLong() == (cursor[domain] ?: 0L)) { "delivery cursor disagreement" }
446:             business += emittedBusiness; if (disposition == "SETTLED") settled++
447:         }
448:         fun requirePrefix(count: Int) {
449:             require(ordinal + 1 == count.toLong()) { "incomplete input prefix" }
450:             if (count == config["inputs"].size()) require(pending.values.all { it.isEmpty() } && owners.values.all { it["continuation"].isNull }) { "financial phase remains unfinished" }
451:         }
452:         private fun actionKey(input: JsonNode) = json.writeValueAsString(listOf("namespace", "domain", "actionId").map { input.path(it) })
453:         private fun requestDigest(input: JsonNode): String {
454:             // Independent source normalization: expected malformed input retains canonical identity while staged.
455:             val normalized = runCatching {
456:                 val payload = input.path("payload")
457:                 require(payload.isObject)
458:                 val numeric = setOf("quantity", "priceNanos", "dueTick", "attempt", "amount", "tick")
459:                 val pairs = json.createArrayNode()
460:                 payload.fieldNames().asSequence().sorted().forEach { name ->
461:                     val value = payload[name]
462:                     val normalizedValue = if (name in numeric) {
463:                         require(value.isTextual && value.asText().matches(Regex("-?[0-9]+")))
464:                         json.nodeFactory.textNode(value.asText().toBigInteger().toString())
465:                     } else value
466:                     pairs.add(json.createArrayNode().add(name).add(normalizedValue))
467:                 }
468:                 json.writeValueAsString(listOf(input.path("kind").asText().uppercase(), input.get("requestedPolicy"), pairs))
469:             }
470:             return sha(normalized.getOrElse { canonical(input) })
471:         }

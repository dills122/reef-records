81:         val key = actionKey(input)
82:         val kind = input.path("kind").asText().uppercase()

84:         val digest = hash(normalized.getOrElse { canonical(input) })

86:         val conflictId = frame(listOf(key, "conflict", digest))
87:         val prior = if (original != null && original["digest"].asText() != digest) state["dedup"][conflictId] ?: original else original
88:         if (prior != null && prior["digest"].asText() == digest) {
89:             if (stagedKey != null) {

91:             }

94:             }
95:         }
96:         val d = Changes()
97:         if (stagedKey != null) { d.put(listOf("stagedInputs", stagedKey), null); d.increment(listOf("deliveryCursor")) }

99:             put("policy", input.path("requestedPolicy").takeUnless { it.isMissingNode || it.isNull }?.asText() ?: state["activePolicy"].asText())

101:         }

104:             prior != null -> "ACTION_CONFLICT"

107:             else -> runCatching {
108:                 when (kind) {
109:                     "CAPTURE" -> capture(input["payload"], d)
110:                     "SETTLE" -> settle(input["payload"], d, legs, false)
111:                     "FUND" -> fund(input["payload"], d, legs)
112:                     "ACTIVATE_POLICY" -> {
113:                         val policy = required(input["payload"], "policy")
114:                         require(policy in setOf("gross-p1", "gross-p2"))

116:                     }
117:                     "CLOCK" -> clock(input, d)

120:                 }

123:                 // Expected invalid input must never retain partial planned mutations.
124:                 return processRejected(input, stagedKey, key, digest, context, if (error is ArithmeticException) "AMOUNT_OVERFLOW" else "INVALID_INPUT", detailed)

177:     private fun settle(p: JsonNode, d: Changes, legs: ArrayNode, continuation: Boolean): String {
178:         val key = resolve(p, if (continuation) "workId" else "executionId")
179:         val execution = state["executions"][key]
180:         val obligation = state["obligations"][key]
181:         val ordinal = if (continuation) "1" else integer(p["attempt"]).also { require(it.signum() > 0) }.toString()
182:         checked(BigInteger(ordinal))
183:         if (continuation) {
184:             require(!state["continuation"].isNull && required(p, "clockAction") == state["continuation"]["clockAction"].asText())
185:             require(dueIndex.firstEntry()?.value == key && BigInteger(state["dueQueue"][key]["tick"].asText()) <= BigInteger(state["logicalTick"].asText()))
186:         }
187:         if (obligation["status"].asText() == "SETTLED") return "ALREADY_SETTLED"
188:         val attemptKey = frame(mapper.readTree(key).map { it.asText() } + ordinal)
189:         if (state["attempts"].has(attemptKey)) return "PRIOR_ATTEMPT"
190:         val cash = BigInteger(obligation["cashResidual"].asText()); val shares = BigInteger(obligation["shareResidual"].asText())
191:         var status = when {
192:             balance("buyerCash") < cash -> "INSUFFICIENT_CASH"
193:             balance("sellerShares") < shares -> "INSUFFICIENT_SHARES"
194:             else -> "SETTLED"
195:         }
196:         if (status == "SETTLED") {
197:             val updates = mapOf("buyerCash" to balance("buyerCash") - cash, "sellerCash" to balance("sellerCash") + cash,
198:                 "buyerShares" to balance("buyerShares") + shares, "sellerShares" to balance("sellerShares") - shares)
199:             if (updates.values.any { !fits(it) }) status = "BALANCE_OVERFLOW" else {
200:                 updates.forEach { (field, amount) -> d.put(listOf("balances", field), text(amount.toString())); d.increment(listOf("versions", field)) }
201:                 legs.add(leg("USD_NANO", "buyer", -cash)); legs.add(leg("USD_NANO", "seller", cash))
202:                 legs.add(leg("ACME_SHARE", "seller", -shares)); legs.add(leg("ACME_SHARE", "buyer", shares))
203:                 d.put(listOf("obligations", key), obj("status" to text("SETTLED"), "cashResidual" to text("0"), "shareResidual" to text("0")))
204:                 d.put(listOf("workflows", key), text("SETTLED")); d.put(listOf("instructions", key), text("SETTLED"))
205:             }
206:         }
207:         d.put(listOf("attempts", attemptKey), text(status)); d.increment(listOf("versions", key))
208:         if (status != "SETTLED") d.put(listOf("exceptions", attemptKey), text(status))
209:         d.put(listOf("dueQueue", key), null)
210:         if (continuation && dueIndex.higherKey(dueIndex.firstKey())?.tick?.let { it <= BigInteger(state["logicalTick"].asText()) } != true) d.put(listOf("continuation"), mapper.nullNode())
211:         return status

314:     private fun evolve(record: JsonNode) {
315:         require(record["sequence"].asLong() == state["historySeq"].asLong() + 1 && record["priorSequence"].asLong() == state["historySeq"].asLong()) { "history hole or reordered suffix" }
316:         require(record["priorChecksum"].asText() == chain) { "history chain mismatch" }
317:         val body = record.deepCopy<ObjectNode>().apply { remove("checksum") }
318:         require(record["checksum"].asText() == hash(canonical(body))) { "history tamper" }
319:         // Validate all preconditions before applying any mutation.
320:         record["changes"].forEach { c -> require(equivalent(at(state, c["path"].map { it.asText() }) ?: mapper.nullNode(), c["before"])) { "delta precondition mismatch" } }
321:         record["changes"].forEach { c ->
322:             val path = c["path"].map { it.asText() }; val after = c["after"]
323:             val parent = at(state, path.dropLast(1)) as ObjectNode
324:             if (after.isNull && path.size > 1) parent.remove(path.last()) else parent.set<JsonNode>(path.last(), if (path.first() == "versions" || (path.size == 1 && path.first() in setOf("businessSeq", "historySeq", "deliveryCursor", "nextDelivery"))) {
325:                 require(after.isIntegralNumber && fits(after.bigIntegerValue()) && after.bigIntegerValue().signum() >= 0)
326:                 number(after.asLong())
327:             } else if (path.first() == "stagedInputs") after.deepCopy<ObjectNode>().apply {
328:                 val position = get("position")
329:                 require(position.isIntegralNumber && fits(position.bigIntegerValue()) && position.bigIntegerValue().signum() >= 0)
330:                 put("position", position.asLong())
331:             } else after.deepCopy<JsonNode>())
332:             if (path.first() == "executions" && !after.isNull) executionIndex.getOrPut(after["executionId"].asText()) { mutableSetOf() }.add(path[1])
333:             if (path.first() == "dueQueue") {
334:                 c["before"].takeUnless { it.isNull }?.let { dueIndex.remove(Due(BigInteger(it["tick"].asText()), it["workId"].asText(), path[1])) }
335:                 if (!after.isNull) dueIndex[Due(BigInteger(after["tick"].asText()), after["workId"].asText(), path[1])] = path[1]
336:             }
337:         }
338:         chain = record["checksum"].asText()
339:         latestRecord = record.deepCopy<JsonNode>()
340:     }
